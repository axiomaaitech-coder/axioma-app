-- ============================================================================
-- PARTE 1 — TESOURARIA_CONFIG (ATIVO)
-- ============================================================================
-- 1 linha por empresa (UNIQUE empresa_id). reserva_minima começa em 0 —
-- o dono ajusta depois; dias_alerta_ruptura em 30 é o horizonte padrão do
-- Radar (mesma janela do Fluxo de Caixa de 13 semanas ~90 dias, mas o
-- ALERTA de ruptura precoce é mais curto que a visão inteira).

CREATE TABLE IF NOT EXISTS public.tesouraria_config (
  id                 uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id         uuid NOT NULL UNIQUE REFERENCES public.empresas(id) ON DELETE CASCADE,
  reserva_minima     numeric(14,2) NOT NULL DEFAULT 0,
  dias_alerta_ruptura int NOT NULL DEFAULT 30,
  criado_em          timestamptz NOT NULL DEFAULT now(),
  atualizado_em      timestamptz NOT NULL DEFAULT now()
);

ALTER TABLE public.tesouraria_config ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS tesouraria_config_empresa ON public.tesouraria_config;
CREATE POLICY tesouraria_config_empresa ON public.tesouraria_config
  FOR ALL TO authenticated
  USING (empresa_id IN (SELECT public.empresas_do_usuario()))
  WITH CHECK (empresa_id IN (SELECT public.empresas_do_usuario()));

-- empresa_id já é UNIQUE (índice implícito) — nenhum índice extra necessário.

-- Trigger: toda empresa nova já nasce com a config padrão — mesmo padrão de
-- fn_empresa_criar_plano_de_contas_padrao (insere direto, sem checar posse:
-- NEW.id é a própria empresa que este INSERT acabou de criar).
CREATE OR REPLACE FUNCTION public.fn_empresa_criar_tesouraria_config_padrao()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  INSERT INTO public.tesouraria_config (empresa_id) VALUES (NEW.id)
  ON CONFLICT (empresa_id) DO NOTHING;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_empresa_criar_tesouraria_config_padrao ON public.empresas;
CREATE TRIGGER trg_empresa_criar_tesouraria_config_padrao
AFTER INSERT ON public.empresas
FOR EACH ROW EXECUTE FUNCTION public.fn_empresa_criar_tesouraria_config_padrao();

-- Backfill — cobre toda empresa que já existia antes deste trigger.
INSERT INTO public.tesouraria_config (empresa_id)
SELECT id FROM public.empresas
ON CONFLICT (empresa_id) DO NOTHING;


-- ============================================================================
-- PARTE 2 — TESOURARIA_CONTA (ATIVO)
-- ============================================================================
-- Marca quais contas do plano são "de tesouraria" (entram na Posição de
-- Caixa). conta_id sem ON DELETE CASCADE/SET NULL não é necessário aqui
-- porque QUEM é dono da linha é esta tabela — CASCADE em conta_id é seguro
-- (se a conta do plano some, a marcação de tesouraria dela não faz sentido
-- sozinha).

CREATE TABLE IF NOT EXISTS public.tesouraria_conta (
  id             uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id     uuid NOT NULL REFERENCES public.empresas(id) ON DELETE CASCADE,
  conta_id       uuid NOT NULL REFERENCES public.plano_de_contas(id) ON DELETE CASCADE,
  tipo_liquidez  text NOT NULL CHECK (tipo_liquidez IN ('disponivel','aplicado','restrito')),
  banco_nome     text,
  ativo          boolean NOT NULL DEFAULT true,
  criado_em      timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT uq_tesouraria_conta UNIQUE (empresa_id, conta_id)
);

ALTER TABLE public.tesouraria_conta ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS tesouraria_conta_empresa ON public.tesouraria_conta;
CREATE POLICY tesouraria_conta_empresa ON public.tesouraria_conta
  FOR ALL TO authenticated
  USING (empresa_id IN (SELECT public.empresas_do_usuario()))
  WITH CHECK (empresa_id IN (SELECT public.empresas_do_usuario()));

CREATE INDEX IF NOT EXISTS idx_tesouraria_conta_empresa ON public.tesouraria_conta (empresa_id);

-- Trigger: toda conta 1.01/1.02/1.03 nova (de QUALQUER empresa, em QUALQUER
-- momento — seed inicial, backfill futuro, ou plano recriado manualmente)
-- já nasce marcada como conta de tesouraria. Ver decisão de desenho no
-- cabeçalho sobre por que isto está em plano_de_contas e não em empresas.
CREATE OR REPLACE FUNCTION public.fn_plano_de_contas_marcar_tesouraria()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF NEW.codigo IN ('1.01', '1.02', '1.03') THEN
    INSERT INTO public.tesouraria_conta (empresa_id, conta_id, tipo_liquidez)
    VALUES (NEW.empresa_id, NEW.id, CASE WHEN NEW.codigo = '1.03' THEN 'aplicado' ELSE 'disponivel' END)
    ON CONFLICT (empresa_id, conta_id) DO NOTHING;
  END IF;
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_plano_de_contas_marcar_tesouraria ON public.plano_de_contas;
CREATE TRIGGER trg_plano_de_contas_marcar_tesouraria
AFTER INSERT ON public.plano_de_contas
FOR EACH ROW EXECUTE FUNCTION public.fn_plano_de_contas_marcar_tesouraria();

-- Backfill — cobre toda empresa que já tinha 1.01/1.02/1.03 antes deste
-- trigger existir.
INSERT INTO public.tesouraria_conta (empresa_id, conta_id, tipo_liquidez)
SELECT empresa_id, id, CASE WHEN codigo = '1.03' THEN 'aplicado' ELSE 'disponivel' END
FROM public.plano_de_contas
WHERE codigo IN ('1.01', '1.02', '1.03')
ON CONFLICT (empresa_id, conta_id) DO NOTHING;

-- ============================================================================
-- PARTE 8 — RPC TESOURARIA_POSICAO (Posição de Caixa)
-- ============================================================================
-- Por conta de tesouraria ativa, saldo até p_data. A soma reaproveita a
-- MESMA regra de saldoNatural() (lib/contabilidadeRelatoriosHelpers.ts):
-- devedora = débito−crédito, credora = crédito−débito. CTE "saldos" agrega
-- as partidas primeiro (uma linha por conta) pra não arrastar o filtro de
-- data pro LEFT JOIN externo por engano — isso é o que garante que uma
-- conta sem nenhum lançamento até p_data ainda aparece com saldo 0, não
-- some da lista.

CREATE OR REPLACE FUNCTION public.tesouraria_posicao(p_empresa_id uuid, p_data date)
RETURNS TABLE (
  conta_id      uuid,
  conta_codigo  text,
  conta_nome    text,
  tipo_liquidez text,
  banco_nome    text,
  saldo         numeric
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF p_empresa_id NOT IN (SELECT public.empresas_do_usuario()) THEN
    RAISE EXCEPTION 'Sem acesso a esta empresa';
  END IF;

  RETURN QUERY
  WITH saldos AS (
    SELECT lcp.conta_id,
      SUM(CASE WHEN lcp.tipo = 'debito'  THEN lcp.valor ELSE 0 END) AS total_debito,
      SUM(CASE WHEN lcp.tipo = 'credito' THEN lcp.valor ELSE 0 END) AS total_credito
    FROM public.lancamento_contabil_partida lcp
    JOIN public.lancamento_contabil lc ON lc.id = lcp.lancamento_id
    WHERE lcp.empresa_id = p_empresa_id AND lc.data <= p_data
    GROUP BY lcp.conta_id
  )
  SELECT
    tc.conta_id,
    pc.codigo,
    pc.nome,
    tc.tipo_liquidez,
    tc.banco_nome,
    CASE WHEN pc.natureza = 'devedora'
      THEN COALESCE(s.total_debito, 0) - COALESCE(s.total_credito, 0)
      ELSE COALESCE(s.total_credito, 0) - COALESCE(s.total_debito, 0)
    END AS saldo
  FROM public.tesouraria_conta tc
  JOIN public.plano_de_contas pc ON pc.id = tc.conta_id
  LEFT JOIN saldos s ON s.conta_id = tc.conta_id
  WHERE tc.empresa_id = p_empresa_id AND tc.ativo;
END;
$$;

REVOKE ALL ON FUNCTION public.tesouraria_posicao(uuid, date) FROM public;
GRANT EXECUTE ON FUNCTION public.tesouraria_posicao(uuid, date) TO authenticated;


-- ============================================================================
-- PARTE 9 — RPC TESOURARIA_FLUXO_PROJETADO (Forecast)
-- ============================================================================
-- Saldo atual (soma de tesouraria_posicao até hoje) + contas_receber em
-- aberto com vencimento até p_dias − contas_pagar em aberto com vencimento
-- até p_dias. "Em aberto" usa a mesma convenção do resto do projeto
-- (status <> 'pago'/'recebido' — ver lib/relatoriosHelpers.ts,
-- lib/contasPagarHelpers.ts).

CREATE OR REPLACE FUNCTION public.tesouraria_fluxo_projetado(p_empresa_id uuid, p_dias int)
RETURNS TABLE (
  saldo_atual         numeric,
  entradas_previstas  numeric,
  saidas_previstas    numeric,
  saldo_projetado     numeric,
  data_limite         date
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_saldo_atual  numeric;
  v_entradas     numeric;
  v_saidas       numeric;
  v_data_limite  date;
BEGIN
  IF p_empresa_id NOT IN (SELECT public.empresas_do_usuario()) THEN
    RAISE EXCEPTION 'Sem acesso a esta empresa';
  END IF;

  v_data_limite := current_date + p_dias;

  SELECT COALESCE(SUM(t.saldo), 0) INTO v_saldo_atual
  FROM public.tesouraria_posicao(p_empresa_id, current_date) t;

  SELECT COALESCE(SUM(valor - COALESCE(valor_recebido, 0)), 0) INTO v_entradas
  FROM public.contas_receber
  WHERE empresa_id = p_empresa_id AND status <> 'recebido' AND data_vencimento <= v_data_limite;

  SELECT COALESCE(SUM(valor_total - COALESCE(valor_pago, 0)), 0) INTO v_saidas
  FROM public.contas_pagar
  WHERE empresa_id = p_empresa_id AND status <> 'pago' AND data_vencimento <= v_data_limite;

  RETURN QUERY SELECT v_saldo_atual, v_entradas, v_saidas, v_saldo_atual + v_entradas - v_saidas, v_data_limite;
END;
$$;

REVOKE ALL ON FUNCTION public.tesouraria_fluxo_projetado(uuid, int) FROM public;
GRANT EXECUTE ON FUNCTION public.tesouraria_fluxo_projetado(uuid, int) TO authenticated;


COMMIT;


-- ============================================================================
-- PARTE 10 — VERIFICAÇÃO (só leitura — rodar depois da transação acima)
-- ============================================================================

SELECT table_name FROM information_schema.tables
  WHERE table_schema = 'public' AND table_name IN (
    'tesouraria_config','tesouraria_conta','tesouraria_politica','tesouraria_cenario',
    'tesouraria_alerta','tesouraria_banco_metrica','tesouraria_forecast_precisao'
  );

SELECT tablename, policyname, cmd FROM pg_policies WHERE tablename LIKE 'tesouraria_%';

SELECT indexname FROM pg_indexes WHERE schemaname = 'public' AND tablename LIKE 'tesouraria_%';

SELECT routine_name FROM information_schema.routines WHERE routine_schema = 'public'
  AND routine_name IN ('tesouraria_posicao','tesouraria_fluxo_projetado',
    'fn_empresa_criar_tesouraria_config_padrao','fn_plano_de_contas_marcar_tesouraria');

-- Toda empresa deveria ter 1 linha de config e 3 contas de tesouraria (1.01/1.02/1.03).
SELECT e.id, e.nome,
  (SELECT count(*) FROM public.tesouraria_config c WHERE c.empresa_id = e.id) AS tem_config,
  (SELECT count(*) FROM public.tesouraria_conta tc WHERE tc.empresa_id = e.id) AS contas_tesouraria
FROM public.empresas e
ORDER BY contas_tesouraria;
