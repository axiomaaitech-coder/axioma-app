-- ============================================================================
-- PARTE 1 — NFE_ITENS: detalhe linha-a-linha da NF-e que o parseXMLNFe já
-- extrai (ItemNFe) e hoje é descartado. Filha de estoque_nfe_importadas.
-- ============================================================================

CREATE TABLE IF NOT EXISTS public.nfe_itens (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id uuid NOT NULL REFERENCES public.empresas(id) ON DELETE CASCADE,
  nfe_importada_id uuid NOT NULL REFERENCES public.estoque_nfe_importadas(id) ON DELETE CASCADE,
  numero_linha integer NOT NULL,
  codigo_fornecedor text,
  ean text,
  descricao text NOT NULL,
  ncm text,
  cfop text,
  unidade text,
  quantidade numeric NOT NULL,
  valor_unitario numeric NOT NULL,
  valor_total numeric NOT NULL,
  numero_lote text,
  data_validade date,
  criado_em timestamptz NOT NULL DEFAULT now()
);

ALTER TABLE public.nfe_itens ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS nfe_itens_select ON public.nfe_itens;

-- ============================================================================
-- PARTE 6 — MATCH_RESULTADO: 1 linha por NF-e conferida — a granularidade
-- da Fila de Exceção (um humano revisa UMA nota).
-- ============================================================================

CREATE TABLE IF NOT EXISTS public.match_resultado (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id uuid NOT NULL REFERENCES public.empresas(id) ON DELETE CASCADE,
  nfe_importada_id uuid NOT NULL REFERENCES public.estoque_nfe_importadas(id) ON DELETE CASCADE,
  contas_pagar_id uuid REFERENCES public.contas_pagar(id) ON DELETE SET NULL,
  nivel text NOT NULL DEFAULT '2way',
  status text NOT NULL DEFAULT 'pendente',
  score integer,
  resumo jsonb,
  decidido_por uuid REFERENCES auth.users(id) ON DELETE SET NULL,
  decidido_em timestamptz,
  criado_em timestamptz NOT NULL DEFAULT now()
);

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'match_resultado_nivel_check'
  ) THEN
    ALTER TABLE public.match_resultado
      ADD CONSTRAINT match_resultado_nivel_check CHECK (nivel IN ('2way', '3way'));
  END IF;
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'match_resultado_status_check'
  ) THEN
    ALTER TABLE public.match_resultado
      ADD CONSTRAINT match_resultado_status_check CHECK (status IN ('pendente', 'ok', 'excecao', 'aprovado', 'rejeitado'));
  END IF;
END $$;

ALTER TABLE public.match_resultado ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS match_resultado_select ON public.match_resultado;
DROP POLICY IF EXISTS match_resultado_insert ON public.match_resultado;
DROP POLICY IF EXISTS match_resultado_update ON public.match_resultado;
DROP POLICY IF EXISTS match_resultado_delete ON public.match_resultado;

-- ============================================================================

  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id uuid NOT NULL REFERENCES public.empresas(id) ON DELETE CASCADE,
  match_resultado_id uuid NOT NULL REFERENCES public.match_resultado(id) ON DELETE CASCADE,
  nfe_item_id uuid REFERENCES public.nfe_itens(id) ON DELETE SET NULL,
  tipo text NOT NULL,
  esperado numeric,
  encontrado numeric,
  descricao_item text,
  criado_em timestamptz NOT NULL DEFAULT now()
);

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint WHERE conname = 'match_divergencias_tipo_check'
  ) THEN
    ALTER TABLE public.match_divergencias

      CHECK (tipo IN ('valor', 'quantidade', 'nao_recebido', 'recebido_sem_nota', 'sem_conta'));

END $$;
