-- Ambas as funções (H1 e H2) recusam se:
--   • quem chama não é dono/admin da empresa dona da movimentação (AX020)

--   • o turno daquela movimentação já está fechado (AX022 — reaproveitado,
--     mesmo código que retaguarda_fechar_turno já usa pra "turno já
--     fechado", mesmo significado aqui)
-- H1 também recusa valor ≤ 0 (AX024 — reaproveitado, mesmo código que

SELECT policyname, cmd FROM pg_policies WHERE schemaname = 'public' AND tablename = 'caixa_movimentacao';


BEGIN;

-- ----------------------------------------------------------------------------
-- H1) retaguarda_editar_movimentacao — corrige valor/motivo de UMA sangria
-- ou suprimento existente. Tipo não muda.
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.retaguarda_editar_movimentacao(
  p_movimentacao_id uuid,
  p_valor numeric,
  p_motivo text DEFAULT NULL
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_empresa_id uuid;
  v_turno_id uuid;
  v_turno_status text;
BEGIN
  SELECT cm.empresa_id, cm.turno_caixa_id INTO v_empresa_id, v_turno_id
  FROM caixa_movimentacao cm WHERE cm.id = p_movimentacao_id;

  IF v_empresa_id IS NULL THEN
    RAISE EXCEPTION 'Movimentação não encontrada.' USING errcode = 'AX028';
  END IF;

  IF public.meu_papel(v_empresa_id) NOT IN ('dono', 'admin') THEN
    RAISE EXCEPTION 'Apenas dono ou administrador podem editar sangria/suprimento.' USING errcode = 'AX020';
  END IF;

  SELECT status INTO v_turno_status FROM turno_caixa WHERE id = v_turno_id;
  IF v_turno_status IS DISTINCT FROM 'aberto' THEN
    RAISE EXCEPTION 'Este turno já está fechado.' USING errcode = 'AX022';
  END IF;

  IF p_valor IS NULL OR p_valor <= 0 THEN
    RAISE EXCEPTION 'Informe um valor válido, maior que zero.' USING errcode = 'AX024';
  END IF;

  UPDATE caixa_movimentacao
  SET valor = p_valor, motivo = p_motivo
  WHERE id = p_movimentacao_id;
END;
$$;

REVOKE ALL ON FUNCTION public.retaguarda_editar_movimentacao(uuid, numeric, text) FROM public;
GRANT EXECUTE ON FUNCTION public.retaguarda_editar_movimentacao(uuid, numeric, text) TO authenticated;

-- ----------------------------------------------------------------------------
-- H2) retaguarda_excluir_movimentacao — apaga UMA sangria ou suprimento
-- (hard delete — registro operacional, não documento fiscal).
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.retaguarda_excluir_movimentacao(p_movimentacao_id uuid)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_empresa_id uuid;
  v_turno_id uuid;
  v_turno_status text;
BEGIN
  SELECT cm.empresa_id, cm.turno_caixa_id INTO v_empresa_id, v_turno_id
  FROM caixa_movimentacao cm WHERE cm.id = p_movimentacao_id;

  IF v_empresa_id IS NULL THEN
    RAISE EXCEPTION 'Movimentação não encontrada.' USING errcode = 'AX028';
  END IF;

  IF public.meu_papel(v_empresa_id) NOT IN ('dono', 'admin') THEN
    RAISE EXCEPTION 'Apenas dono ou administrador podem excluir sangria/suprimento.' USING errcode = 'AX020';
  END IF;

  SELECT status INTO v_turno_status FROM turno_caixa WHERE id = v_turno_id;
  IF v_turno_status IS DISTINCT FROM 'aberto' THEN
    RAISE EXCEPTION 'Este turno já está fechado.' USING errcode = 'AX022';
  END IF;

  DELETE FROM caixa_movimentacao WHERE id = p_movimentacao_id;
END;
$$;

REVOKE ALL ON FUNCTION public.retaguarda_excluir_movimentacao(uuid) FROM public;
GRANT EXECUTE ON FUNCTION public.retaguarda_excluir_movimentacao(uuid) TO authenticated;

-- ----------------------------------------------------------------------------
-- H3) RLS de caixa_movimentacao ganha UPDATE e DELETE — mesma condição do
-- INSERT (caixa_movimentacao_escrita, Fase 1): dono/admin da empresa dona
-- da movimentação. Defesa em profundidade — H1/H2 são SECURITY DEFINER e
-- não dependem desta política pra funcionar.
-- ----------------------------------------------------------------------------
DROP POLICY IF EXISTS caixa_movimentacao_atualizacao ON public.caixa_movimentacao;
CREATE POLICY caixa_movimentacao_atualizacao ON public.caixa_movimentacao
  FOR UPDATE TO authenticated
  USING (
    empresa_id IN (SELECT public.empresas_do_usuario())
    AND public.meu_papel(empresa_id) IN ('dono', 'admin')
  )
  WITH CHECK (
    empresa_id IN (SELECT public.empresas_do_usuario())
    AND public.meu_papel(empresa_id) IN ('dono', 'admin')
  );

DROP POLICY IF EXISTS caixa_movimentacao_exclusao ON public.caixa_movimentacao;
CREATE POLICY caixa_movimentacao_exclusao ON public.caixa_movimentacao
  FOR DELETE TO authenticated
  USING (
    empresa_id IN (SELECT public.empresas_do_usuario())
    AND public.meu_papel(empresa_id) IN ('dono', 'admin')
  );

COMMENT ON TABLE public.caixa_movimentacao IS
  'Sangria (retirada) e suprimento (reforço) de dinheiro no caixa físico durante o turno — só dono/admin. Criar via retaguarda_registrar_movimentacao(), editar via retaguarda_editar_movimentacao(), excluir via retaguarda_excluir_movimentacao() (só com o turno aberto). Ver PDV-RETAGUARDA-SQL-REVISAO.txt e PDV-RETAGUARDA-FASE4-EDITAR-EXCLUIR-MOV-SQL.txt.';

COMMIT;
