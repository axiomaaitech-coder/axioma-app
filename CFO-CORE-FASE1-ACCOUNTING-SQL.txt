-- (AP_CREATED/AP_PAID/SALE_CREATED/...) é o próximo commit, depois que este

-- Nota à parte: já existe uma tabela "lancamentos_centro" (planilha manual

-- ORDEM DE EXECUÇÃO (transação única — se qualquer parte falhar, nada fica
-- pela metade):
--   PASSO 0  — leitura: confirma as 4 peças-base acima existem
--   PARTE 1  — tabela plano_de_contas (gerencial, com de-para SPED)
--   PARTE 2  — RLS de plano_de_contas (1 policy FOR ALL — o dono edita)
--   PARTE 3  — índices de plano_de_contas
--   PARTE 4  — tabela lancamento_contabil (cabeçalho, imutável)
--   PARTE 5  — RLS de lancamento_contabil (SELECT+INSERT — igual eventos_negocio)
--   PARTE 6  — índices de lancamento_contabil
--   PARTE 7  — tabela lancamento_contabil_partida (linhas, imutável)
--   PARTE 8  — RLS de lancamento_contabil_partida (SELECT+INSERT)
--   PARTE 9  — índices de lancamento_contabil_partida
--   PARTE 10 — TRIGGER DE PARTIDA DOBRADA (última linha de defesa — ver nota)
--   PARTE 11 — verificação (só leitura)
--   ROLLBACK (comentado, só referência)
--
-- Idempotente (IF NOT EXISTS / DROP POLICY|TRIGGER IF EXISTS antes de
-- recriar) — pode rodar mais de uma vez com segurança.
--
-- DECISÃO DE DESENHO — PARTIDA DOBRADA EM CAMADAS (nunca uma defesa única):
--   Camada 1 (principal): função pura TypeScript que soma débito/crédito de
--   um array de partidas ANTES de qualquer insert — testável isoladamente,
--   sem banco, e é o que dá o feedback rápido no formulário.
--   Camada 2 (última linha, nunca a única): CONSTRAINT TRIGGER no Postgres
--   (Parte 10) — DEFERRABLE INITIALLY DEFERRED, dispara por linha inserida
--   mas só EXECUTA de fato no COMMIT da transação, depois que todas as
--   linhas daquele lançamento já foram inseridas. Nesse momento soma
--   SUM(valor) WHERE tipo='debito' e SUM(valor) WHERE tipo='credito' pro
--   mesmo lancamento_id; se não baterem, RAISE EXCEPTION aborta a transação
--   inteira (nenhuma linha fica gravada pela metade). Não dá pra fazer isso
--   com CHECK constraint comum porque CHECK só enxerga a própria linha, não
--   soma entre linhas — por isso precisa ser trigger. A regra de negócio
--   "isto é uma venda, debita Caixa credita Receita" nunca mora aqui — o
--   trigger só verifica que a MATEMÁTICA fecha, é cego ao significado.
--
-- DECISÃO DE DESENHO — NATUREZA é campo independente de TIPO (não derivado).
--   Na maioria dos casos tipo determina a natureza (ativo/despesa=devedora,
--   passivo/patrimonio/receita=credora), mas CONTAS REDUTORAS invertem isso
--   por desenho contábil (ex: "Depreciação Acumulada" é tipo=ativo mas
--   natureza=credora; "Devoluções de Vendas" é tipo=receita mas
--   natureza=devedora). Se natureza fosse derivada de tipo, não teria como
--   representar essas contas — por isso fica explícita, cadastrada por
--   quem monta o plano.
--
-- DECISÃO DE DESENHO — imposto (IBS/CBS/ICMS/PIS/COFINS): nenhum enum ou
--   CHECK list de nomes de conta existe neste schema — codigo/nome são texto
--   livre. Isso já garante que o plano comporte tanto contas de imposto
--   legadas (ICMS/PIS/COFINS) quanto as da Reforma Tributária (IBS/CBS) sem
--   nenhuma migração futura — a estrutura não trava em nenhum dos dois
--   regimes. Nenhuma conta de imposto é criada neste commit (ver Diagnóstico
--   3 na resposta, sobre o seed).
--
-- DECISÃO DE DESENHO — estornado_por_id é o ÚNICO ponteiro de reversão
--   (auto-FK, no lançamento ORIGINAL, apontando pro lançamento NOVO que o
--   reverteu). "O que esse lançamento reverteu" é a pergunta inversa (WHERE
--   estornado_por_id = :id), coberta por índice — não precisa de 2ª coluna.
--   Como lancamento_contabil só tem RLS de SELECT+INSERT (nunca UPDATE), a
--   escrita de estornado_por_id no original PRECISA vir de uma função
--   SECURITY DEFINER dedicada — isso é lógica de estorno, fica pro próximo

-- ============================================================================
-- PARTE 1 — TABELA PLANO_DE_CONTAS
-- ============================================================================
-- Plano GERENCIAL (o dono usa — livre, enxuto). codigo_referencial_rfb é o
-- de-para pro Plano Referencial da Receita/SPED, preenchido quando o
-- contador do cliente mapear cada conta — nullable porque no dia 1 (Simples

-- conta_pai_id monta a hierarquia (ex: "1.1.01 Caixa" filha de "1.1 Ativo

CREATE TABLE IF NOT EXISTS public.plano_de_contas (
  id                     uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id             uuid NOT NULL REFERENCES public.empresas(id) ON DELETE CASCADE,
  codigo                 text NOT NULL,
  nome                   text NOT NULL,
  tipo                   text NOT NULL CHECK (tipo IN ('ativo','passivo','patrimonio','receita','despesa')),
  natureza               text NOT NULL CHECK (natureza IN ('devedora','credora')),
  conta_pai_id           uuid REFERENCES public.plano_de_contas(id) ON DELETE SET NULL,
  codigo_referencial_rfb text,
  ativo                  boolean NOT NULL DEFAULT true,
  criado_em              timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT uq_plano_de_contas_empresa_codigo UNIQUE (empresa_id, codigo)
);

  tipo            text NOT NULL CHECK (tipo IN ('debito','credito')),
  valor           numeric(14,2) NOT NULL CHECK (valor > 0),
  centro_custo_id uuid REFERENCES public.centros_custo(id) ON DELETE SET NULL,
  criado_em       timestamptz NOT NULL DEFAULT now()

-- ============================================================================
-- PARTE 10 — TRIGGER DE PARTIDA DOBRADA (última linha de defesa)
-- ============================================================================
-- CONSTRAINT TRIGGER, não trigger comum: DEFERRABLE INITIALLY DEFERRED faz
-- o Postgres só EXECUTAR isto no COMMIT da transação, não a cada linha
-- inserida — essencial, porque um lançamento com 2+ linhas não fecha até a
-- ÚLTIMA ser inserida (checar linha por linha rejeitaria a 1ª inserção
-- sempre, já que sozinha ela nunca bate débito=crédito). Isso exige que o
-- código que grava um lançamento insira TODAS as suas linhas na MESMA
-- transação (uma chamada de RPC, não múltiplos requests soltos) — é a
-- responsabilidade que o próximo commit (a função de gravar lançamento)
-- precisa cumprir pra este trigger fazer sentido.
--
-- SECURITY DEFINER + search_path fixo: mesmo padrão de
-- fn_contas_pagar_auditoria_trigger, evita qualquer surpresa de RLS na
-- consulta interna de soma (a policy de SELECT já libera isto de qualquer
-- forma, já que é a própria empresa lendo sua própria linha — SECURITY
-- DEFINER aqui é consistência de padrão, não uma necessidade estrita).

CREATE OR REPLACE FUNCTION public.fn_checar_partida_dobrada()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_debito   numeric;
  v_credito  numeric;
BEGIN
  SELECT
    COALESCE(SUM(valor) FILTER (WHERE tipo = 'debito'), 0),
    COALESCE(SUM(valor) FILTER (WHERE tipo = 'credito'), 0)
  INTO v_debito, v_credito
  FROM public.lancamento_contabil_partida
  WHERE lancamento_id = NEW.lancamento_id;

  IF v_debito <> v_credito THEN
    RAISE EXCEPTION 'Partida não fechada no lançamento %: débito=% crédito=%', NEW.lancamento_id, v_debito, v_credito;
  END IF;

  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_partida_dobrada ON public.lancamento_contabil_partida;
CREATE CONSTRAINT TRIGGER trg_partida_dobrada
AFTER INSERT ON public.lancamento_contabil_partida
DEFERRABLE INITIALLY DEFERRED
FOR EACH ROW EXECUTE FUNCTION public.fn_checar_partida_dobrada();

REVOKE ALL ON FUNCTION public.fn_checar_partida_dobrada() FROM public;


COMMIT;
