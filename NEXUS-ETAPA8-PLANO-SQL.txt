-- ============================================================================
-- AXIOMA NEXUS — ETAPA 8: PLANO DO JOSÉ PARA A EMPRESA (tabela nexus_plano_empresa)
-- Rodar UMA VEZ no SQL Editor do Supabase. Idempotente. Não mexe em nada existente.
-- Plano de sobrevivência e crescimento por empresa × horizonte (1-3, 4-7, 8-10
-- anos) × idioma × dia. Dado PRIVADO da empresa: empresa_id + RLS por
-- empresas_do_usuario() (mesmo padrão de nexus_simulation / multi-tenant).
-- ============================================================================

BEGIN;

CREATE TABLE IF NOT EXISTS public.nexus_plano_empresa (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  empresa_id  uuid NOT NULL REFERENCES public.empresas(id) ON DELETE CASCADE,
  horizonte   text NOT NULL CHECK (horizonte IN ('1-3','4-7','8-10')),
  lang        text NOT NULL CHECK (lang IN ('pt','en','es')),
  data        date NOT NULL,
  conteudo    jsonb NOT NULL,
  modelo      text,
  criado_por  uuid REFERENCES auth.users(id) ON DELETE SET NULL,
  gerado_em   timestamptz NOT NULL DEFAULT now(),
  UNIQUE (empresa_id, horizonte, lang, data)
);

ALTER TABLE public.nexus_plano_empresa ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS nexus_plano_empresa_multitenant ON public.nexus_plano_empresa;
CREATE POLICY nexus_plano_empresa_multitenant ON public.nexus_plano_empresa
  FOR ALL TO authenticated
  USING (empresa_id IN (SELECT public.empresas_do_usuario()))
  WITH CHECK (empresa_id IN (SELECT public.empresas_do_usuario()));

CREATE INDEX IF NOT EXISTS idx_nexus_plano_empresa_busca
  ON public.nexus_plano_empresa (empresa_id, horizonte, lang, data DESC);

COMMIT;
