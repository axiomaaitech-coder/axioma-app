-- ----------------------------------------------------------------------------
-- 5.1) ap_detectar_duplicata — mesmo fornecedor + valor ±1% dentro da janela,
-- OU mesmo número de nota (não vazio). Score: 60 (só janela+valor), +25 se
-- numero_nota bate, +15 se a data de emissão é a mesma dia. Empresa vem do
-- contexto (auth.uid() via empresas_do_usuario()), nunca aceita de fora —
-- evita um usuário pedir duplicata de empresa alheia.
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.ap_detectar_duplicata(
  p_empresa_id uuid,
  p_fornecedor_id uuid,
  p_valor_total numeric,
  p_data_emissao date,
  p_numero_nota text DEFAULT NULL,
  p_dias_janela integer DEFAULT 30
)
RETURNS TABLE (
  contas_pagar_id uuid,
  descricao text,
  numero_nota text,
  valor_total numeric,
  data_emissao date,
  data_vencimento date,
  score integer
)
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF p_empresa_id NOT IN (SELECT public.empresas_do_usuario()) THEN
    RAISE EXCEPTION 'Sem acesso a esta empresa';
  END IF;

  RETURN QUERY
  SELECT
    cp.id, cp.descricao, cp.numero_nota, cp.valor_total, cp.data_emissao, cp.data_vencimento,
    (
      60
      + CASE WHEN p_numero_nota IS NOT NULL AND p_numero_nota <> '' AND cp.numero_nota = p_numero_nota THEN 25 ELSE 0 END
      + CASE WHEN cp.data_emissao = p_data_emissao THEN 15 ELSE 0 END
    )::integer AS score
  FROM public.contas_pagar cp
  WHERE cp.empresa_id = p_empresa_id
    AND cp.fornecedor_id IS NOT DISTINCT FROM p_fornecedor_id
    AND cp.data_emissao IS NOT NULL
    AND cp.data_emissao BETWEEN (p_data_emissao - (p_dias_janela || ' days')::interval) AND (p_data_emissao + (p_dias_janela || ' days')::interval)
    AND (
      (p_valor_total > 0 AND ABS(cp.valor_total - p_valor_total) <= (p_valor_total * 0.01))
      OR (p_numero_nota IS NOT NULL AND p_numero_nota <> '' AND cp.numero_nota = p_numero_nota)
    )
  ORDER BY score DESC;
END;
$$;

REVOKE ALL ON FUNCTION public.ap_detectar_duplicata(uuid, uuid, numeric, date, text, integer) FROM public;
GRANT EXECUTE ON FUNCTION public.ap_detectar_duplicata(uuid, uuid, numeric, date, text, integer) TO authenticated;


-- ----------------------------------------------------------------------------
-- 5.2) ap_registrar_auditoria — helper interno, chamado pelo trigger (Parte 6)
-- E diretamente pelo client pra eventos que não são INSERT/UPDATE/DELETE de
-- verdade (duplicata_detectada, duplicata_ignorada, aprovou, rejeitou).
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.ap_registrar_auditoria(
  p_contas_pagar_id uuid,
  p_acao text,
  p_antes jsonb DEFAULT NULL,
  p_depois jsonb DEFAULT NULL
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_empresa_id uuid;
  v_id uuid;
BEGIN
  SELECT empresa_id INTO v_empresa_id FROM public.contas_pagar WHERE id = p_contas_pagar_id;
  IF v_empresa_id IS NULL THEN
    RAISE EXCEPTION 'Conta a pagar % não encontrada', p_contas_pagar_id;
  END IF;
  IF v_empresa_id NOT IN (SELECT public.empresas_do_usuario()) THEN
    RAISE EXCEPTION 'Sem acesso a esta empresa';
  END IF;

  INSERT INTO public.contas_pagar_auditoria (contas_pagar_id, empresa_id, usuario_id, acao, antes, depois)
  VALUES (p_contas_pagar_id, v_empresa_id, auth.uid(), p_acao, p_antes, p_depois)
  RETURNING id INTO v_id;

  RETURN v_id;
END;
$$;

REVOKE ALL ON FUNCTION public.ap_registrar_auditoria(uuid, text, jsonb, jsonb) FROM public;
GRANT EXECUTE ON FUNCTION public.ap_registrar_auditoria(uuid, text, jsonb, jsonb) TO authenticated;


-- ----------------------------------------------------------------------------
-- 5.3) ap_solicitar_aprovacao — se valor > limite, cria aprovação pendente E
-- marca a conta como 'aguardando_aprovacao' (trava o pagamento até decidir —
-- ver darBaixaContaPagar no Commit 4, que passa a checar esse status). Se
-- dentro do limite, marca auto_aprovada e não mexe no status da conta.
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.ap_solicitar_aprovacao(p_contas_pagar_id uuid)
RETURNS TABLE (aprovacao_id uuid, status text)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_empresa_id uuid;
  v_valor numeric;
  v_limite numeric;
  v_status text;
  v_id uuid;
BEGIN
  SELECT empresa_id, valor_total INTO v_empresa_id, v_valor FROM public.contas_pagar WHERE id = p_contas_pagar_id;
  IF v_empresa_id IS NULL THEN
    RAISE EXCEPTION 'Conta a pagar % não encontrada', p_contas_pagar_id;
  END IF;
  IF v_empresa_id NOT IN (SELECT public.empresas_do_usuario()) THEN
    RAISE EXCEPTION 'Sem acesso a esta empresa';
  END IF;

  SELECT limite_aprovacao_automatica INTO v_limite FROM public.empresa_config_ap WHERE empresa_id = v_empresa_id;
  v_limite := COALESCE(v_limite, 500); -- empresa sem linha de config ainda = usa o padrão (500)

  IF v_valor > v_limite THEN
    v_status := 'pendente';
  ELSE
    v_status := 'auto_aprovada';
  END IF;

  INSERT INTO public.contas_pagar_aprovacao (contas_pagar_id, empresa_id, solicitante_id, valor, status, decidido_em)
  VALUES (p_contas_pagar_id, v_empresa_id, auth.uid(), v_valor, v_status, CASE WHEN v_status = 'auto_aprovada' THEN now() ELSE NULL END)
  RETURNING id INTO v_id;

  IF v_status = 'pendente' THEN
    UPDATE public.contas_pagar SET status = 'aguardando_aprovacao' WHERE id = p_contas_pagar_id;
  END IF;

  RETURN QUERY SELECT v_id, v_status;
END;
$$;

REVOKE ALL ON FUNCTION public.ap_solicitar_aprovacao(uuid) FROM public;
GRANT EXECUTE ON FUNCTION public.ap_solicitar_aprovacao(uuid) TO authenticated;


-- ----------------------------------------------------------------------------
-- 5.4) ap_decidir_aprovacao — só quem está em empresa_config_ap.aprovadores
-- OU é o dono da empresa (fail-safe: lista de aprovadores vazia nunca deixa
-- a empresa sem ninguém que possa decidir). Ao aprovar, devolve a conta pro
-- status calculado por valor/pago/vencimento (o client recalcula e chama
-- editarContaPagar — esta função só desbloqueia, não decide o status final,
-- porque só o app sabe se já foi paga/vencida).
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.ap_decidir_aprovacao(
  p_aprovacao_id uuid,
  p_decisao text,
  p_motivo text DEFAULT NULL
)
RETURNS TABLE (contas_pagar_id uuid, status text)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_empresa_id uuid;
  v_contas_pagar_id uuid;
  v_dono_id uuid;
  v_aprovadores uuid[];
  v_pode_decidir boolean;
BEGIN
  IF p_decisao NOT IN ('aprovada', 'rejeitada') THEN
    RAISE EXCEPTION 'Decisão inválida: %', p_decisao;
  END IF;

  SELECT empresa_id, contas_pagar_id INTO v_empresa_id, v_contas_pagar_id
  FROM public.contas_pagar_aprovacao WHERE id = p_aprovacao_id AND status = 'pendente';
  IF v_empresa_id IS NULL THEN
    RAISE EXCEPTION 'Aprovação % não encontrada ou já decidida', p_aprovacao_id;
  END IF;
  IF v_empresa_id NOT IN (SELECT public.empresas_do_usuario()) THEN
    RAISE EXCEPTION 'Sem acesso a esta empresa';
  END IF;

  SELECT user_id INTO v_dono_id FROM public.empresas WHERE id = v_empresa_id;
  SELECT aprovadores INTO v_aprovadores FROM public.empresa_config_ap WHERE empresa_id = v_empresa_id;
  v_pode_decidir := (auth.uid() = v_dono_id) OR (v_aprovadores IS NOT NULL AND auth.uid() = ANY(v_aprovadores));

  IF NOT v_pode_decidir THEN
    RAISE EXCEPTION 'Você não está habilitado a aprovar contas desta empresa';
  END IF;

  UPDATE public.contas_pagar_aprovacao
  SET status = p_decisao, motivo = p_motivo, aprovador_id = auth.uid(), decidido_em = now()
  WHERE id = p_aprovacao_id;

  -- rejeitada: volta pro estado anterior (pendente/vencido, recalculado pelo
  -- client). aprovada: sai de 'aguardando_aprovacao', também recalculado
  -- pelo client (calcStatus já existe em fornecedorHelpers.ts — não duplicado
  -- aqui em SQL).
  UPDATE public.contas_pagar SET status = 'pendente' WHERE id = v_contas_pagar_id AND status = 'aguardando_aprovacao';

  PERFORM public.ap_registrar_auditoria(v_contas_pagar_id, CASE WHEN p_decisao = 'aprovada' THEN 'aprovou' ELSE 'rejeitou' END, NULL, jsonb_build_object('motivo', p_motivo));

  RETURN QUERY SELECT v_contas_pagar_id, 'pendente'::text;
END;
$$;

REVOKE ALL ON FUNCTION public.ap_decidir_aprovacao(uuid, text, text) FROM public;
GRANT EXECUTE ON FUNCTION public.ap_decidir_aprovacao(uuid, text, text) TO authenticated;


-- ============================================================================
-- PARTE 6 — TRIGGER DE AUDITORIA AUTOMÁTICA em contas_pagar
-- ============================================================================
-- Cobre criou/editou/excluiu mesmo se algum caminho de código futuro
-- esquecer de registrar manualmente (a Fase 1 do Fluxo de Caixa/Fornecedores
-- não chama nada disso — o trigger garante que nada escapa, sem depender de
-- toda tela lembrar de chamar um helper).

CREATE OR REPLACE FUNCTION public.fn_contas_pagar_auditoria_trigger()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    INSERT INTO public.contas_pagar_auditoria (contas_pagar_id, empresa_id, usuario_id, acao, antes, depois)
    VALUES (NEW.id, NEW.empresa_id, auth.uid(), 'criou', NULL, to_jsonb(NEW));
    RETURN NEW;
  ELSIF TG_OP = 'UPDATE' THEN
    INSERT INTO public.contas_pagar_auditoria (contas_pagar_id, empresa_id, usuario_id, acao, antes, depois)
    VALUES (
      NEW.id, NEW.empresa_id, auth.uid(),
      CASE
        WHEN OLD.status <> 'pago' AND NEW.status = 'pago' THEN 'baixou'
        WHEN OLD.status = 'pago' AND NEW.status <> 'pago' THEN 'estornou'
        ELSE 'editou'
      END,
      to_jsonb(OLD), to_jsonb(NEW)
    );
    RETURN NEW;
  ELSIF TG_OP = 'DELETE' THEN
    INSERT INTO public.contas_pagar_auditoria (contas_pagar_id, empresa_id, usuario_id, acao, antes, depois)
    VALUES (OLD.id, OLD.empresa_id, auth.uid(), 'excluiu', to_jsonb(OLD), NULL);
    RETURN OLD;
  END IF;
  RETURN NULL;
END;
$$;

DROP TRIGGER IF EXISTS trg_contas_pagar_auditoria ON public.contas_pagar;
CREATE TRIGGER trg_contas_pagar_auditoria
AFTER INSERT OR UPDATE OR DELETE ON public.contas_pagar
FOR EACH ROW EXECUTE FUNCTION public.fn_contas_pagar_auditoria_trigger();
