--    livre, já testado em produção).
--    Sem seed/backfill de atividade_fiscal por adivinhação (evidence-first) —
--    não dá pra inferir comércio/indústria/serviço de dado nenhum hoje
--    disponível sem chutar. Fica NULL até o dono preencher (Rodada 2/tela).

-- fiscal_descoberta, fiscal_health, fiscal_regra_versao,
-- fiscal_agente_execucao: NENHUMA tabela equivalente encontrada — construídas

-- ============================================================================
-- ORDEM DE EXECUÇÃO (transação única — se qualquer parte falhar, nada fica
-- pela metade):
--   PASSO 0   — leitura: confirma as peças-base existem
--   PARTE 1    — ALTER empresas (atividade_fiscal, aliquota_iss_pct) [ATIVO]
--   PARTE 2    — ALTER empresa_obrigacoes (competencia, risco, evidencia,

--   PARTE 3-5  — fiscal_descoberta (tabela, RLS, índices)            [ATIVO]
--   PARTE 6-7  — fiscal_health (tabela, RLS, índice)                 [ATIVO]
--   PARTE 8-9  — fiscal_regra_versao (tabela, RLS)                   [INERTE — gancho pra Regulatory Intelligence, ninguém grava/lê ainda]
--   PARTE 10-11 — fiscal_agente_execucao (tabela, RLS)                [INERTE — gancho pra ZIA/Copilot fiscal, ninguém grava ainda]

--   PARTE 13   — RPC fiscal_obrigacoes_proximas
--   PARTE 14   — verificação (só leitura)
--   ROLLBACK (comentado, só referência)
--
-- Idempotente (ADD COLUMN/CREATE ... IF NOT EXISTS, DROP POLICY|FUNCTION IF

--
-- ============================================================================
-- DECISÕES DE DESENHO
-- ============================================================================
--
-- RLS — mesmos 3 padrões do Contador, mesmo motivo:
--   1) fiscal_descoberta: FOR ALL. Tabela "viva" — o dono lê e age (marcar

--   2) fiscal_health: SELECT apenas. 100% calculada — só a RPC

--      fato consumado, mesmo princípio de contador_agente_execucao.
--   fiscal_regra_versao: SELECT apenas pra authenticated (regra pode ser
--      global — empresa_id NULL — ou por empresa). Nenhuma policy de

--      versionamento — regra nunca é sobrescrita, sempre nova linha).
--
-- Por que fiscal_descoberta é tabela PRÓPRIA e não contador_descoberta

--   gargalo de 2 motores de regras crescendo em paralelo, e mantém a fronteira
--   de módulo clara (mesma razão de mei_obrigacoes ≠ empresa_obrigacoes).
--   Custo aceito: 2 tabelas quase-idênticas. Trade-off registrado, não
--   escondido.
--
-- competencia (empresa_obrigacoes, fiscal_health) como DATE, convenção
--   SEMPRE dia 1 do mês — mesma escolha e mesmo motivo já documentado no
--   Contador (RPCs fazem aritmética de data). mei_obrigacoes continua com
--   competencia TEXT porque só faz igualdade/agrupamento, nunca range —
--   divergência conhecida, não é bug.
--
-- Índice parcial em vez de UNIQUE direto em empresa_obrigacoes(empresa_id,
--   tipo, competencia): dados manuais já existentes (anos de uso da aba
--   Obrigações) podem ter duplicidade de tipo+competência sem problema
--   nenhum pro usuário (ex.: 2 lançamentos "ICMS" no mesmo mês). Forçar
--   UNIQUE em cima disso quebraria a migração se existir 1 linha assim. A
--   uniqueness só precisa valer pro upsert idempotente do MOTOR Fiscal
--   (linhas com origem <> 'manual') — por isso o índice é parcial
--   (WHERE origem <> 'manual'), nunca toca dado manual existente.
--
-- exposicao (fiscal_health) = soma de valor_estimado das obrigações
--   pendentes/atrasadas da competência — é o número que já existe em
--   empresa_obrigacoes.valor_estimado, sem inventar cálculo novo.
--
-- documentos_pendentes (fiscal_health) fica 0 nesta rodada — a fonte real
--   (NF-e/Open Finance) é integração futura, ainda não wired pro Fiscal.
--   Não inventamos contagem sem fonte (evidence-first). Campo já existe no
--   schema pra não exigir migração quando a integração ligar.
--
-- qualidade_dados (fiscal_health) = % de campos-chave preenchidos em
--   empresas (regime_tributario, cnae_principal, atividade_fiscal, e
--   aliquota_iss_pct SE atividade_fiscal indicar serviço/misto) — mede
--   literalmente "dá pra calcular imposto certo com o que está cadastrado

-- score (fiscal_health) 0-1000 = média ponderada (50% em-dia, 30% qualidade

--
-- Nenhum enum/CHECK de nome de imposto ou obrigação: mesmo motivo já

  ADD COLUMN IF NOT EXISTS aliquota_iss_pct numeric CHECK (aliquota_iss_pct IS NULL OR (aliquota_iss_pct >= 0 AND aliquota_iss_pct <= 100));

-- Sem seed/backfill — ver "DECISÕES DE DESENHO" (evidence-first, não chuta

  id                uuid PRIMARY KEY DEFAULT gen_random_uuid(),

  tipo              text NOT NULL CHECK (tipo IN ('inconsistencia','anomalia','oportunidade','risco','divergencia','concentracao','classificacao_suspeita','tendencia')),
  prioridade        text NOT NULL CHECK (prioridade IN ('P0','P1','P2','P3')),

  causa             text,

  confianca         text NOT NULL CHECK (confianca IN ('fato','calculo','inferencia','previsao','cenario')),

-- ============================================================================
-- PARTE 12 — RPC FISCAL_HEALTH_CALCULAR                                [ATIVO]
-- ============================================================================
-- Score 0-1000 = 50% em-dia + 30% qualidade de dados + 20% ausência de
-- divergências abertas (cap 10) — ver "DECISÕES DE DESENHO". Determinístico,
-- upsert em fiscal_health por (empresa_id, competencia).

CREATE OR REPLACE FUNCTION public.fiscal_health_calcular(
  p_empresa_id  uuid,
  p_competencia date
)
RETURNS SETOF public.fiscal_health
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_competencia_inicio date := date_trunc('month', p_competencia)::date;
  v_competencia_fim    date := (date_trunc('month', p_competencia) + interval '1 month' - interval '1 day')::date;
  v_obrig_total        int;
  v_obrig_em_dia       int;
  v_pct_em_dia         numeric;
  v_exposicao          numeric;
  v_divergencias       int;
  v_campos_ok          numeric;
  v_campos_esperados   numeric;
  v_qualidade          numeric;
  v_score              numeric;
  v_detalhe            jsonb;
BEGIN
  IF p_empresa_id NOT IN (SELECT public.empresas_do_usuario()) THEN
    RAISE EXCEPTION 'Sem acesso a esta empresa';
  END IF;

  -- 1) Obrigações em dia na competência (data_vencimento no mês, ou
  -- competencia == mês pra linhas já geradas pelo motor).
  SELECT COUNT(*),
    COUNT(*) FILTER (WHERE status IN ('paga','dispensada','confirmada','enviada') OR (status = 'pendente' AND data_vencimento >= CURRENT_DATE)),
    COALESCE(SUM(valor_estimado) FILTER (WHERE status IN ('pendente','atrasada')), 0)
  INTO v_obrig_total, v_obrig_em_dia, v_exposicao
  FROM public.empresa_obrigacoes
  WHERE empresa_id = p_empresa_id
    AND (
      (competencia IS NOT NULL AND competencia = v_competencia_inicio)
      OR (competencia IS NULL AND data_vencimento BETWEEN v_competencia_inicio AND v_competencia_fim)
    );

  v_pct_em_dia := CASE WHEN v_obrig_total = 0 THEN 100 ELSE ROUND(v_obrig_em_dia::numeric / v_obrig_total * 100, 1) END;

  -- 2) Divergências abertas (fiscal_descoberta), sem filtro de período —
  -- ver "DECISÕES DE DESENHO".
  SELECT COUNT(*) INTO v_divergencias
  FROM public.fiscal_descoberta
  WHERE empresa_id = p_empresa_id AND status = 'aberto';

  -- 3) Qualidade de dados: % de campos-chave preenchidos pra calcular
  -- imposto certo.
  SELECT
    (CASE WHEN regime_tributario IS NOT NULL AND regime_tributario <> '' THEN 1 ELSE 0 END) +
    (CASE WHEN cnae_principal IS NOT NULL AND cnae_principal <> '' THEN 1 ELSE 0 END) +
    (CASE WHEN atividade_fiscal IS NOT NULL THEN 1 ELSE 0 END) +
    (CASE WHEN atividade_fiscal NOT IN ('servico','misto') THEN 1 ELSE (CASE WHEN aliquota_iss_pct IS NOT NULL THEN 1 ELSE 0 END) END)
  INTO v_campos_ok
  FROM public.empresas WHERE id = p_empresa_id;

  v_campos_esperados := 4;
  v_qualidade := ROUND(COALESCE(v_campos_ok, 0) / v_campos_esperados * 100, 1);

  v_score := ROUND(
    (v_pct_em_dia * 0.5) +
    (v_qualidade * 0.3) +
    (GREATEST(0, 100 - LEAST(v_divergencias, 10) * 10) * 0.2)
  * 10);

  v_detalhe := jsonb_build_object(
    'formula', '50% em_dia + 30% qualidade_dados + 20% (100 - min(divergencias,10)*10), escala x10 -> 0-1000',
    'obrigacoes_em_dia_pct', v_pct_em_dia, 'obrigacoes_total', v_obrig_total, 'obrigacoes_em_dia', v_obrig_em_dia,
    'qualidade_dados_pct', v_qualidade, 'campos_ok', v_campos_ok, 'campos_esperados', v_campos_esperados,
    'divergencias_abertas', v_divergencias
  );

  RETURN QUERY
  INSERT INTO public.fiscal_health (empresa_id, competencia, score, obrigacoes_em_dia_pct, divergencias, documentos_pendentes, exposicao, qualidade_dados, detalhe, calculado_em)
  VALUES (p_empresa_id, v_competencia_inicio, v_score, v_pct_em_dia, v_divergencias, 0, v_exposicao, v_qualidade, v_detalhe, now())
  ON CONFLICT (empresa_id, competencia) DO UPDATE SET
    score                 = EXCLUDED.score,
    obrigacoes_em_dia_pct = EXCLUDED.obrigacoes_em_dia_pct,
    divergencias          = EXCLUDED.divergencias,
    documentos_pendentes  = EXCLUDED.documentos_pendentes,
    exposicao             = EXCLUDED.exposicao,
    qualidade_dados       = EXCLUDED.qualidade_dados,
    detalhe               = EXCLUDED.detalhe,
    calculado_em          = now()
  RETURNING *;
END;
$$;

REVOKE ALL ON FUNCTION public.fiscal_health_calcular(uuid, date) FROM public;
GRANT EXECUTE ON FUNCTION public.fiscal_health_calcular(uuid, date) TO authenticated;


-- ============================================================================
-- PARTE 13 — RPC FISCAL_OBRIGACOES_PROXIMAS                            [ATIVO]
-- ============================================================================
-- Lista obrigações vencendo nos próximos p_dias, com status e o que falta.
-- Nunca devolve linha já 'paga'/'dispensada'.

CREATE OR REPLACE FUNCTION public.fiscal_obrigacoes_proximas(
  p_empresa_id uuid,
  p_dias       int DEFAULT 30
)
RETURNS TABLE (
  id              uuid,
  tipo            text,
  nome            text,
  data_vencimento date,
  status          text,
  risco           text,
  valor_estimado  numeric,
  dias_restantes  int
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF p_empresa_id NOT IN (SELECT public.empresas_do_usuario()) THEN
    RAISE EXCEPTION 'Sem acesso a esta empresa';
  END IF;

  RETURN QUERY
  SELECT eo.id, eo.tipo, eo.nome, eo.data_vencimento, eo.status, eo.risco, eo.valor_estimado,
    (eo.data_vencimento - CURRENT_DATE)::int AS dias_restantes
  FROM public.empresa_obrigacoes eo
  WHERE eo.empresa_id = p_empresa_id
    AND eo.status NOT IN ('paga', 'dispensada')
    AND eo.data_vencimento BETWEEN CURRENT_DATE AND (CURRENT_DATE + make_interval(days => p_dias))
  ORDER BY eo.data_vencimento ASC;
END;
$$;

REVOKE ALL ON FUNCTION public.fiscal_obrigacoes_proximas(uuid, int) FROM public;
GRANT EXECUTE ON FUNCTION public.fiscal_obrigacoes_proximas(uuid, int) TO authenticated;


COMMIT;
