-- PARTE D — CORREÇÃO DE SEGURANÇA: item_venda fecha pro operador
-- (a lacuna do item 3 que você pediu pra corrigir agora)
--
-- O PROBLEMA: item_venda guarda preco_unitario_venda E
-- custo_unitario_na_venda por linha vendida. Hoje a política dela
-- (item_venda_multi_tenant, criada em PDV-FASE3-ETAPA1-VENDAS-SQL.sql) usa
-- empresas_do_usuario() — dá acesso total a QUALQUER papel da empresa,
-- inclusive operador, porque na época a intenção era "operador precisa
-- registrar/ver a própria venda". RLS do Postgres bloqueia por LINHA, não
-- por COLUNA — então um operador que soubesse chamar a API do Supabase
-- direto (fora da tela) conseguiria ler custo_unitario_na_venda de
-- qualquer venda da empresa, mesmo a tela nunca mostrando isso.
--
-- POR QUE NÃO É "VIEW SEGURA" (o padrão de vw_produtos_seguro) SOZINHA:
-- vw_produtos_seguro funciona porque a tabela produtos JÁ bloqueia 100% das
-- linhas pro operador (RLS de PDV-FASE0), então a view é o ÚNICO caminho de
-- leitura que sobra pra ele — e ela simplesmente não seleciona as colunas
-- sensíveis. Item_venda é diferente: a tabela crua está ABERTA pro operador
-- hoje. Criar uma view segura do lado, sem fechar a tabela crua, não
-- resolve nada — o operador continuaria conseguindo ler
-- custo_unitario_na_venda direto da tabela, ignorando a view.
--
-- POR QUE NÃO É "REVOKE de coluna" (GRANT/REVOKE por coluna do Postgres):
-- documentado no cabeçalho de PDV-FASE3-ETAPA0-OPERADOR-PRODUTOS-SQL.sql —
-- todo usuário autenticado do Axioma (dono, operador, admin...) é o MESMO
-- role do Postgres ("authenticated"). REVOKE SELECT (coluna) FROM
-- authenticated tiraria o acesso de TODO MUNDO, incluindo dono/admin/
-- financeiro — não dá pra diferenciar papel por essa via aqui.
--
-- A ABORDAGEM ESCOLHIDA (a mais robusta disponível nesta arquitetura):
-- trocar a política de item_venda de empresas_do_usuario() (acesso total,
-- todo papel) pra empresas_do_usuario_operacional() (bloqueia por LINHA
-- quem tem vínculo 'operador') — o MESMO padrão que toda outra tabela
-- sensível do projeto já usa (é literalmente a política PADRÃO da
-- descoberta dinâmica da Fase 0; item_venda só não a tinha porque foi
-- colocada na lista de exceções por precaução, não por necessidade real
-- comprovada). Com isso, operador passa a ter ZERO linhas em item_venda —
-- nem custo, nem preço, nem nada — exatamente como já acontece hoje com
-- produtos.
--
-- POR QUE ISSO NÃO QUEBRA O PDV (verifiquei antes de propor, não é
-- suposição): busquei todo o repositório por uso de "item_venda" no código
-- do app. Resultado: A ÚNICA referência em todo o projeto é um COUNT (sem
-- selecionar nenhuma coluna, `head: true`) em lib/estoqueHelpers.ts, dentro
-- de excluirProduto() — usado só pra decidir se um produto pode ser
-- excluído ou precisa ser inativado, chamado só nas telas de gestão de
-- estoque/catálogo (já bloqueadas pro operador antes mesmo de chegar
-- nisso). NINGUÉM no código lê item_venda de volta pra montar carrinho,
-- recibo ou qualquer tela do operador — o cupom de impressão (Fase 3,
-- Etapa 3) usa o estado do carrinho no navegador, nunca reconsulta o
-- banco. E a gravação em si (finalizar_venda) é SECURITY DEFINER — ela
-- grava em item_venda com o próprio privilégio dela, nunca depende da RLS
-- de quem chamou. Ou seja: apertar esta política não tira NENHUM acesso
-- que o operador realmente usa hoje.
--
-- Depois desta mudança, item_venda passa a se comportar exatamente como
-- produtos: se um dia alguma tela precisar que o operador releia linhas de
-- venda (ex.: reimprimir cupom buscando do banco em vez do estado local),
-- aí sim cria-se uma vw_item_venda_seguro (mesmo molde de
-- vw_produtos_seguro) — não faço isso agora porque nenhum código precisa
-- disso hoje (YAGNI).
-- ============================================================================

DROP POLICY IF EXISTS item_venda_multi_tenant ON public.item_venda;
CREATE POLICY item_venda_multi_tenant ON public.item_venda
  FOR ALL TO authenticated
  USING (empresa_id IN (SELECT public.empresas_do_usuario_operacional()))
  WITH CHECK (empresa_id IN (SELECT public.empresas_do_usuario_operacional()));

COMMENT ON POLICY item_venda_multi_tenant ON public.item_venda IS
  'Bloqueia por LINHA quem tem vínculo papel=operador (empresas_do_usuario_operacional) — fecha o acesso a custo_unitario_na_venda pro operador. Trocado de empresas_do_usuario() em PDV-RETAGUARDA-SQL-REVISAO.txt (ver comentário completo ali); finalizar_venda() continua gravando normalmente (SECURITY DEFINER, ignora esta política).';

-- ----------------------------------------------------------------------------
-- E1) retaguarda_resumo_dia — números do dia (painel ao vivo E fechamento
-- usam o mesmo resumo diário; só a conferência de gaveta é por turno).
-- Se retaguarda_ver_lucro = false, lucro_real volta NULL — filtrado aqui
-- dentro, nunca no payload calculado e escondido só na tela.
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.retaguarda_resumo_dia(p_empresa_id uuid, p_data date DEFAULT CURRENT_DATE)
RETURNS TABLE (
  total_vendido numeric,
  qtd_vendas integer,
  ticket_medio numeric,
  lucro_real numeric,
  itens_sem_custo integer
)
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_ver_lucro boolean;
BEGIN
  IF public.meu_papel(p_empresa_id) NOT IN ('dono', 'admin') THEN
    RAISE EXCEPTION 'Apenas dono ou administrador podem ver a retaguarda.' USING errcode = 'AX020';
  END IF;

  SELECT retaguarda_ver_lucro INTO v_ver_lucro FROM empresas WHERE id = p_empresa_id;

  RETURN QUERY
  SELECT
    COALESCE(SUM(v.valor_total), 0) AS total_vendido,
    COUNT(v.id)::integer AS qtd_vendas,
    CASE WHEN COUNT(v.id) > 0 THEN ROUND(SUM(v.valor_total) / COUNT(v.id), 2) ELSE 0 END AS ticket_medio,
    CASE WHEN COALESCE(v_ver_lucro, true) THEN (
      SELECT COALESCE(SUM((iv.preco_unitario_venda - COALESCE(iv.custo_unitario_na_venda, 0)) * iv.quantidade), 0)
      FROM item_venda iv
      JOIN venda v2 ON v2.id = iv.venda_id
      WHERE v2.empresa_id = p_empresa_id AND v2.status = 'finalizada' AND v2.finalizada_em::date = p_data
    ) ELSE NULL END AS lucro_real,
    (
      SELECT COUNT(*)::integer
      FROM item_venda iv
      JOIN venda v3 ON v3.id = iv.venda_id
      WHERE v3.empresa_id = p_empresa_id AND v3.status = 'finalizada' AND v3.finalizada_em::date = p_data
        AND iv.custo_unitario_na_venda IS NULL
    ) AS itens_sem_custo
  FROM venda v
  WHERE v.empresa_id = p_empresa_id AND v.status = 'finalizada' AND v.finalizada_em::date = p_data;
END;
$$;

REVOKE ALL ON FUNCTION public.retaguarda_resumo_dia(uuid, date) FROM public;
GRANT EXECUTE ON FUNCTION public.retaguarda_resumo_dia(uuid, date) TO authenticated;

-- ----------------------------------------------------------------------------
-- E2) retaguarda_vendas_por_categoria — agrupado nicho→categoria→sub-nicho
-- (segmento→categoria→subcategoria em produtos).
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.retaguarda_vendas_por_categoria(p_empresa_id uuid, p_data date DEFAULT CURRENT_DATE)
RETURNS TABLE (
  nicho text,
  categoria text,
  sub_nicho text,
  quantidade numeric,
  valor_vendido numeric,
  lucro_real numeric
)
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_ver_lucro boolean;
BEGIN
  IF public.meu_papel(p_empresa_id) NOT IN ('dono', 'admin') THEN
    RAISE EXCEPTION 'Apenas dono ou administrador podem ver a retaguarda.' USING errcode = 'AX020';
  END IF;

  SELECT retaguarda_ver_lucro INTO v_ver_lucro FROM empresas WHERE id = p_empresa_id;

  RETURN QUERY
  SELECT
    COALESCE(NULLIF(p.segmento, ''), '—') AS nicho,
    COALESCE(NULLIF(p.categoria, ''), '—') AS categoria,
    COALESCE(NULLIF(p.subcategoria, ''), '—') AS sub_nicho,
    SUM(iv.quantidade) AS quantidade,
    SUM(iv.subtotal) AS valor_vendido,
    CASE WHEN COALESCE(v_ver_lucro, true)
      THEN SUM((iv.preco_unitario_venda - COALESCE(iv.custo_unitario_na_venda, 0)) * iv.quantidade)
      ELSE NULL END AS lucro_real
  FROM item_venda iv
  JOIN venda v ON v.id = iv.venda_id
  JOIN produtos p ON p.id = iv.produto_id
  WHERE v.empresa_id = p_empresa_id AND v.status = 'finalizada' AND v.finalizada_em::date = p_data
  GROUP BY 1, 2, 3
  ORDER BY valor_vendido DESC;
END;
$$;

REVOKE ALL ON FUNCTION public.retaguarda_vendas_por_categoria(uuid, date) FROM public;
GRANT EXECUTE ON FUNCTION public.retaguarda_vendas_por_categoria(uuid, date) TO authenticated;

-- ----------------------------------------------------------------------------
-- E3) retaguarda_itens_prejuizo — alerta de item vendido abaixo do custo
-- (linha a linha, não agrupado). Extra implícito no plano aprovado ("alerta
-- de item vendido com prejuízo" no painel ao vivo) — sinalizando aqui que
-- não estava nos 4 nomes que você listou, adicionei por ser parte do
-- escopo já aprovado. Some inteira (0 linhas) se ver_lucro = false, porque
-- o alerta em si já revela a relação preço×custo do item.
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.retaguarda_itens_prejuizo(p_empresa_id uuid, p_data date DEFAULT CURRENT_DATE)
RETURNS TABLE (
  venda_id uuid,
  produto_nome text,
  quantidade numeric,
  preco_unitario_venda numeric,
  custo_unitario_na_venda numeric,
  prejuizo_unitario numeric
)
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_ver_lucro boolean;
BEGIN
  IF public.meu_papel(p_empresa_id) NOT IN ('dono', 'admin') THEN
    RAISE EXCEPTION 'Apenas dono ou administrador podem ver a retaguarda.' USING errcode = 'AX020';
  END IF;

  SELECT retaguarda_ver_lucro INTO v_ver_lucro FROM empresas WHERE id = p_empresa_id;
  IF NOT COALESCE(v_ver_lucro, true) THEN
    RETURN; -- config pede pra não mostrar lucro/custo — o alerta some junto
  END IF;

  RETURN QUERY
  SELECT v.id, p.nome, iv.quantidade, iv.preco_unitario_venda, iv.custo_unitario_na_venda,
         (iv.custo_unitario_na_venda - iv.preco_unitario_venda)
  FROM item_venda iv
  JOIN venda v ON v.id = iv.venda_id
  JOIN produtos p ON p.id = iv.produto_id
  WHERE v.empresa_id = p_empresa_id AND v.status = 'finalizada' AND v.finalizada_em::date = p_data
    AND iv.custo_unitario_na_venda IS NOT NULL
    AND iv.preco_unitario_venda < iv.custo_unitario_na_venda
  ORDER BY (iv.custo_unitario_na_venda - iv.preco_unitario_venda) * iv.quantidade DESC;
END;
$$;

REVOKE ALL ON FUNCTION public.retaguarda_itens_prejuizo(uuid, date) FROM public;
GRANT EXECUTE ON FUNCTION public.retaguarda_itens_prejuizo(uuid, date) TO authenticated;

-- ----------------------------------------------------------------------------
-- E4) retaguarda_fechar_turno — calcula o esperado, grava o contado, fecha
-- o turno (status/valor_fechamento/valor_esperado/fechado_em). Se a
-- conferência estiver ligada, exige valor contado; se estiver desligada,
-- fecha sem exigir (mas o esperado é sempre calculado e guardado, pra
-- ficar registrado mesmo que ninguém tenha conferido naquele dia).
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.retaguarda_fechar_turno(
  p_turno_caixa_id uuid,
  p_valor_contado numeric DEFAULT NULL,
  p_observacao text DEFAULT NULL
)
RETURNS TABLE (
  turno_id uuid,
  valor_esperado numeric,
  valor_contado numeric,
  diferenca numeric
)
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_empresa_id uuid;
  v_abertura numeric;
  v_status text;
  v_conferir boolean;
  v_vendas_dinheiro numeric;
  v_suprimentos numeric;
  v_sangrias numeric;
  v_esperado numeric;
BEGIN
  SELECT empresa_id, valor_abertura, status INTO v_empresa_id, v_abertura, v_status
  FROM turno_caixa WHERE id = p_turno_caixa_id;

  IF v_empresa_id IS NULL THEN
    RAISE EXCEPTION 'Turno de caixa não encontrado.' USING errcode = 'AX021';
  END IF;

  IF public.meu_papel(v_empresa_id) NOT IN ('dono', 'admin') THEN
    RAISE EXCEPTION 'Apenas dono ou administrador podem fechar o caixa.' USING errcode = 'AX020';
  END IF;

  IF v_status <> 'aberto' THEN
    RAISE EXCEPTION 'Este turno já está fechado.' USING errcode = 'AX022';
  END IF;

  SELECT retaguarda_conferir_gaveta INTO v_conferir FROM empresas WHERE id = v_empresa_id;
  IF COALESCE(v_conferir, true) AND p_valor_contado IS NULL THEN
    RAISE EXCEPTION 'Informe o valor contado na gaveta para fechar o caixa.' USING errcode = 'AX027';
  END IF;

  SELECT COALESCE(SUM(valor_total), 0) INTO v_vendas_dinheiro
  FROM venda WHERE turno_caixa_id = p_turno_caixa_id AND status = 'finalizada' AND forma_pagamento = 'dinheiro';

  SELECT COALESCE(SUM(valor), 0) INTO v_suprimentos
  FROM caixa_movimentacao WHERE turno_caixa_id = p_turno_caixa_id AND tipo = 'suprimento';

  SELECT COALESCE(SUM(valor), 0) INTO v_sangrias
  FROM caixa_movimentacao WHERE turno_caixa_id = p_turno_caixa_id AND tipo = 'sangria';

  v_esperado := v_abertura + v_vendas_dinheiro + v_suprimentos - v_sangrias;

  UPDATE turno_caixa
  SET status = 'fechado',
      valor_fechamento = p_valor_contado,
      valor_esperado = v_esperado,
      fechado_em = now(),
      observacao = COALESCE(p_observacao, observacao),
      updated_at = now()
  WHERE id = p_turno_caixa_id;

  RETURN QUERY SELECT p_turno_caixa_id, v_esperado, p_valor_contado,
    CASE WHEN p_valor_contado IS NOT NULL THEN p_valor_contado - v_esperado ELSE NULL END;
END;
$$;

REVOKE ALL ON FUNCTION public.retaguarda_fechar_turno(uuid, numeric, text) FROM public;
GRANT EXECUTE ON FUNCTION public.retaguarda_fechar_turno(uuid, numeric, text) TO authenticated;

-- ----------------------------------------------------------------------------
-- E5) retaguarda_registrar_movimentacao — sangria/suprimento, só com
-- conferência ligada e turno aberto.
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.retaguarda_registrar_movimentacao(
  p_turno_caixa_id uuid,
  p_tipo text,
  p_valor numeric,
  p_motivo text DEFAULT NULL
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
  v_empresa_id uuid;
  v_status text;
  v_conferir boolean;
  v_id uuid;
BEGIN
  SELECT empresa_id, status INTO v_empresa_id, v_status FROM turno_caixa WHERE id = p_turno_caixa_id;

  IF v_empresa_id IS NULL THEN
    RAISE EXCEPTION 'Turno de caixa não encontrado.' USING errcode = 'AX021';
  END IF;

  IF public.meu_papel(v_empresa_id) NOT IN ('dono', 'admin') THEN
    RAISE EXCEPTION 'Apenas dono ou administrador podem registrar sangria/suprimento.' USING errcode = 'AX020';
  END IF;

  IF v_status <> 'aberto' THEN
    RAISE EXCEPTION 'Este turno já está fechado.' USING errcode = 'AX022';
  END IF;

  IF p_tipo NOT IN ('sangria', 'suprimento') THEN
    RAISE EXCEPTION 'Tipo de movimentação inválido.' USING errcode = 'AX023';
  END IF;

  IF p_valor IS NULL OR p_valor <= 0 THEN
    RAISE EXCEPTION 'Informe um valor válido, maior que zero.' USING errcode = 'AX024';
  END IF;

  SELECT retaguarda_conferir_gaveta INTO v_conferir FROM empresas WHERE id = v_empresa_id;
  IF NOT COALESCE(v_conferir, true) THEN
    RAISE EXCEPTION 'A conferência de gaveta está desligada nas configurações da retaguarda.' USING errcode = 'AX025';
  END IF;

  INSERT INTO caixa_movimentacao (empresa_id, turno_caixa_id, tipo, valor, motivo, usuario_id)
  VALUES (v_empresa_id, p_turno_caixa_id, p_tipo, p_valor, p_motivo, auth.uid())
  RETURNING id INTO v_id;

  RETURN v_id;
END;
$$;

REVOKE ALL ON FUNCTION public.retaguarda_registrar_movimentacao(uuid, text, numeric, text) FROM public;
GRANT EXECUTE ON FUNCTION public.retaguarda_registrar_movimentacao(uuid, text, numeric, text) TO authenticated;

-- ----------------------------------------------------------------------------
-- E6) retaguarda_salvar_config — grava os 3 botões de configuração. Passa
-- por função própria (em vez do UPDATE direto em empresas que a tela
-- /empresa usa) porque a RLS de UPDATE em empresas hoje é só pra 'dono'
-- (FIX-POLITICA-EMPRESAS-SQL.sql) — e a decisão 2 pede dono E admin
-- podendo configurar a retaguarda. Alargar a RLS de UPDATE de empresas
-- inteira pra admin abriria edição de CNPJ/razão social/endereço etc. pra
-- admin também, que não foi pedido — esta função abre só estas 3 colunas.
-- ----------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.retaguarda_salvar_config(
  p_empresa_id uuid,
  p_modo text,
  p_conferir_gaveta boolean,
  p_ver_lucro boolean
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
BEGIN
  IF public.meu_papel(p_empresa_id) NOT IN ('dono', 'admin') THEN
    RAISE EXCEPTION 'Apenas dono ou administrador podem configurar a retaguarda.' USING errcode = 'AX020';
  END IF;

  IF p_modo NOT IN ('ao_vivo', 'fechamento', 'ambos') THEN
    RAISE EXCEPTION 'Modo de retaguarda inválido.' USING errcode = 'AX026';
  END IF;

  UPDATE empresas
  SET retaguarda_modo = p_modo,
      retaguarda_conferir_gaveta = p_conferir_gaveta,
      retaguarda_ver_lucro = p_ver_lucro
  WHERE id = p_empresa_id;
END;
$$;

REVOKE ALL ON FUNCTION public.retaguarda_salvar_config(uuid, text, boolean, boolean) FROM public;
GRANT EXECUTE ON FUNCTION public.retaguarda_salvar_config(uuid, text, boolean, boolean) TO authenticated;

COMMIT;
